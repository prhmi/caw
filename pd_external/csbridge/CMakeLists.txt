cmake_minimum_required(VERSION 3.15)
project(csbridge C)
# Set PD_INCLUDE to the folder holding m_pd.h (Pd install: ...\src) and PD_LIB to pd.lib (Windows: ...\bin\pd.lib)
set(PD_INCLUDE "C:/Program Files/Pd/src" CACHE PATH "folder with m_pd.h")
set(PD_LIB "C:/Program Files/Pd/bin/pd.lib" CACHE FILEPATH "pd import library")
set(CSOUND_INCLUDE "C:/Program Files/Csound6_x64/include" CACHE PATH "folder containing csound/ or csound.h")
set(CSOUND_LIB "C:/Program Files/Csound6_x64/lib/csound64.lib" CACHE FILEPATH "csound import library")
add_library(csbridge_tilde MODULE "csbridge~.c")
set_target_properties(csbridge_tilde PROPERTIES OUTPUT_NAME "csbridge~" PREFIX "" SUFFIX ".dll")
target_include_directories(csbridge_tilde PRIVATE ${PD_INCLUDE} ${CSOUND_INCLUDE} ${CSOUND_INCLUDE}/..)
target_link_libraries(csbridge_tilde PRIVATE ${PD_LIB} ${CSOUND_LIB})
